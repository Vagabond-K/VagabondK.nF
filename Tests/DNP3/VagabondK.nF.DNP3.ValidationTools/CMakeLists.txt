cmake_minimum_required(VERSION 3.20)
project(VagabondKDnp3Master LANGUAGES CXX)
option(VAGABONDK_NF_PATCHED_IEEE "Use the explicitly labelled test-only IEEE754 replacement" OFF)
if(NOT EXISTS "${OPENDNP3_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Set OPENDNP3_SOURCE to the pinned OpenDNP3 3.1.2 checkout.")
endif()
set(DNP3_TLS OFF CACHE BOOL "" FORCE)
set(DNP3_DOTNET OFF CACHE BOOL "" FORCE)
set(DNP3_EXAMPLES OFF CACHE BOOL "" FORCE)
set(DNP3_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPENDNP3_SOURCE}" opendnp3)
if(VAGABONDK_NF_PATCHED_IEEE)
    target_include_directories(ser4cpp BEFORE INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/ieee-overlay")
    add_compile_definitions(VAGABONDK_NF_PATCHED_IEEE=1)
    add_executable(VagabondKDnp3Decoder decoder.cpp)
    target_link_libraries(VagabondKDnp3Decoder PRIVATE ser4cpp)
    target_compile_features(VagabondKDnp3Decoder PRIVATE cxx_std_14)
endif()
add_executable(VagabondKDnp3Master main.cpp)
target_link_libraries(VagabondKDnp3Master PRIVATE opendnp3)
target_compile_features(VagabondKDnp3Master PRIVATE cxx_std_14)
add_executable(VagabondKDnp3Outstation outstation.cpp)
target_link_libraries(VagabondKDnp3Outstation PRIVATE opendnp3)
target_compile_features(VagabondKDnp3Outstation PRIVATE cxx_std_14)
